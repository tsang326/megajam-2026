# The short route: from the wreck straight to the dock by the yellow camera, plus R.
#   Tools/run_offscreen.sh route_yellow Saved/DeepCut/route_yellow
skipfilm
wait 1.0
goto 3
givepart
wait 1.5
snap y1_wreck_marks
# without the cell: the dock cannot be powered, R goes back
swimmark 0 0
wait 0.4
snap y2_on_yellow_cell_stays
cut
wait 0.8
snap y3_dock_no_cell
retake
wait 0.8
snap y4_back_in_wreck
# push the cell to the right until it sits on the DOCK frame, rise clear of it, stand on the yellow mark
swimrel cell -150 130
swimrel cell -110 0
swimsock dock -70 21 1150
swimrel cell -70 130
swimmark 0 0
wait 0.5
snap y5_on_yellow_cell_fits
cut
wait 0.6
snap y6_power_on
waitstate EndCard
wait 1.0
snap y7_end_card_no_crew
stat
quit
