# The short route: from the wreck straight to the dock by the yellow camera, plus R twice.
#   Tools/run_offscreen.sh route_yellow Saved/DeepCut/route_yellow
skipfilm
wait 1.0
goto 3
givepart
wait 1.5
snap y1_wreck_marks
# without the cell: the dock cannot be powered, R goes back
swimto 300 -50 1150
wait 0.4
snap y2_on_yellow_no_cell
cut
wait 0.8
snap y3_dock_no_cell
retake
wait 0.8
snap y4_back_in_wreck
swimto 380 -130 1250
use
wait 0.4
swimto 300 -50 1150
wait 0.5
snap y5_on_yellow_with_cell
cut
wait 0.6
swimto -150 -40 1250
swimto -260 -40 1250
wait 0.7
use
wait 0.6
snap y6_power_on
waitstate EndCard
wait 1.0
snap y7_end_card_no_crew
stat
quit
